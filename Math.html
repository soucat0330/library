namespace Prime {
  //@brief エラトステネスの篩
  //@param n この整数以下の素数を列挙する
  //@return 素数が格納されたvector
  std::vector<int> sieve(int n) {
    std::vector<bool> is_p(n + 1, true);
    is_p[0] = is_p[1] = false;
    for (int i = 2; i * i < n + 1; i++) {
      if (is_p[i]) {
        for (int j = i * i; j < n + 1; j += i) {
          is_p[j] = false;
        }
      }
    }
    std::vector<int> out;
    for (int i = 1; i < n + 1; i++) {
      if (is_p[i]) {
        out.push_back(i);
      }
    }
    return out;
  }

  //@brief 素因数分解
  //@param n 素因数分解をする対象の整数
  //@return 素因数と指数のペアが素因数の昇順で格納されたベクターを返す
  std::vector<std::pair<int, int>>  PF(int n) {
    std::vector<std::pair<int, int>> out;
    int x = n;
    int count = 0;
    while (x % 2 == 0) {
      count++;
      x /= 2;
    }
    if (count != 0) {
      out.push_back({ 2, count });
    }
    for (int i = 3; i * i <= n; i += 2) {
      count = 0;
      while (x % i == 0) {
        count++;
        x /= i;
      }
      if (count != 0) {
        out.push_back({ i, count });
      }
      if (x == 1) {
        break;
      }
    }
    if (x != 1)out.push_back({ x, 1 });
    return out;
  }
}
